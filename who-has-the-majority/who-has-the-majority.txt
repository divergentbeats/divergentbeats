class Solution {
  public:
    int moreFrequent(vector<int>& arr, int x, int y) {
        // code here
        vector<int> freq(2,0);
        for(int i=0;i<arr.size();i++)
        {
            if(arr[i]== x)
            freq[0]++;
            else if(arr[i]==y)
            freq[1]++;
        }
        if(freq[0]<freq[1])
        return y;
        else if(freq[0]>freq[1])
        return x;
        return min(x,y);
            
    }
};
