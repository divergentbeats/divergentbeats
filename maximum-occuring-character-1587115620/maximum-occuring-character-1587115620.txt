class Solution {
  public:
    char getMaxOccuringChar(string& s) {
        map<char,int> mp;
        for(int i=0;i<s.size();i++)
        mp[s[i]]++;
        int maxcount=0;
        char maxletter;
        for(auto x:mp)
        {
            if(x.second>maxcount)
            {
                maxcount=x.second;
                maxletter=x.first;
            }
        }
        
        return maxletter;
        
    }
};
