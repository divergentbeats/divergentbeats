class Solution {
  public:
    vector<int> getFloorAndCeil(int x, vector<int> &arr) {
        // code here
        sort(arr.begin(),arr.end());
        int floor=-1,ceil=-1;
        
        for(int i=0;i<arr.size();i++)
        {
            if(arr[i]<=x)
            floor=arr[i];
            if(arr[i]>=x && ceil==-1)
            ceil=arr[i];
        }
        return {floor,ceil};
    }
};
