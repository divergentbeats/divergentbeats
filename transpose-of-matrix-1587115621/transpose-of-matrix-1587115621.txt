class Solution {
  public:
    vector<vector<int>> transpose(vector<vector<int>>& mat) {
        int n=mat.size();
        vector<vector<int>> ans(n,vector<int>(n));
        for(int i=0;i<mat.size();i++)
        {
            for(int j=0;j<mat[0].size();j++)
            {
                ans[i][j]=mat[j][i];
            }
        }
        
        return ans;
    }
};
