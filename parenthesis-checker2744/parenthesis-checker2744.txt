class Solution {
  public:
    bool isBalanced(string& s) {
        stack<char> st;
        int i=0;
        while(i<s.size())
        {
            if(!st.empty() && st.top()=='{' && s[i]=='}')
            st.pop();
            else if(!st.empty() && st.top()=='[' && s[i]==']')
            st.pop();
            else if(!st.empty() && st.top()=='(' && s[i]==')')
            st.pop();
            else
            st.push(s[i]);
            i++;
        }
        if(st.empty())
        return true;
        else
        return false;
        
        
    }
};
