class Solution {
  public:
    vector<int> leaders(vector<int>& arr) {
        int n=arr.size();
        vector<int> ans;
        int maxright=arr[n-1];
        ans.push_back(arr[n-1]);
        for(int i=n-2;i>=0;i--)
        {
            if(arr[i]>=maxright)
            {
                maxright=arr[i];
                ans.push_back(arr[i]);
            }
            
        }
        reverse(ans.begin(),ans.end());
        
        
        return ans;

        
    }
};
