class Solution {
  public:
    int findKRotation(vector<int> &arr) {
        int left=0, right=arr.size()-1;
        while(left<=right)
        {
            int mid=(left+right)/2;
            if(arr[mid]<=arr[left] && arr[mid-1]>arr[mid])
            return mid;
            else if(arr[mid]<arr[right])
            right=mid-1;
            else
            left=mid+1;
        }
        return 0;
        
    }
};
