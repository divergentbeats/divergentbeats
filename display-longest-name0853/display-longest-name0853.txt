class Solution {
  public:
    string longest(vector<string>& arr) {
        string word="";
        int size=0;
        for(int i=0;i<arr.size();i++)
        {
            if(arr[i].size()>size)
            {
                word=arr[i];
                size=arr[i].size();
            }
        }
        
        return word;
    }
};
