class Solution {
  public:
    int minRow(vector<vector<int>> &mat) {
        int minrow=INT_MAX,rowcount=INT_MAX;
        for(int i=0;i<mat.size();i++)
        {
            int count=0;
            for(int j=0;j<mat[i].size();j++)
            {
                if(mat[i][j]==1)
                count++;
                
            }
            if(count<rowcount)
            {
                rowcount=count;
                minrow=i+1;
            }
            
        }
        return minrow;
        
    }
};
