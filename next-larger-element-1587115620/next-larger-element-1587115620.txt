class Solution {
public:
    vector<int> nextLargerElement(vector<int>& arr) {
        int n = arr.size();

        vector<int> ans(n);
        stack<int> st;

        for (int i = n - 1; i >= 0; i--) {

            // Remove elements that cannot be the answer
            while (!st.empty() && st.top() <= arr[i]) {
                st.pop();
            }

            // The remaining top is the next greater element
            if (st.empty()) {
                ans[i] = -1;
            } else {
                ans[i] = st.top();
            }

            // Current element becomes a candidate for elements to its left
            st.push(arr[i]);
        }

        return ans;
    }
};
