class Solution {
  public:
    string removeDuplicates(string &s) {
        string ans="";
        unordered_map<char,int> mp;
        for(int i=0;i<s.size();i++)
        {
            if(mp[s[i]]==0)
            {
                ans+=s[i];
                mp[s[i]]++;
            }
        }
        
        return ans;
        
    }
};
