class Solution {
  public:
    void mergeArrays(vector<int>& a, vector<int>& b) {
        for(int i=0;i<b.size();i++)
        {
            a.push_back(b[i]);
        }
        sort(a.begin(),a.end());
        int n=b.size();
        int count=0;
        int i=a.size()-1, j=0;
        while(count<n)
        {
            b[j]=a[i];
            a.pop_back();
            count++;
            j++, i--;
            
        }
        sort(b.begin(),b.end());
    }
};
