class Solution {
  public:
    bool areAnagrams(string& s1, string& s2) {
        if(s1.size()<s2.size() || s2.size()<s1.size())
        return false;
        
        unordered_map<char,int> mp;
        for(int i=0;i<s1.size();i++)
        mp[s1[i]]++;
        for(int i=0;i<s2.size();i++)
        {
            mp[s2[i]]--;
            if(mp[s2[i]]==0)
            mp.erase(s2[i]);
        }
        
        if(mp.empty())
        return true;
        else
        return false;
        
        
    }
};
