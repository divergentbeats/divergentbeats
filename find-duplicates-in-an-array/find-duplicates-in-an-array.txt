class Solution {
  public:
    vector<int> findDuplicates(vector<int>& arr) {
        vector<int> ans;
        map<int,int> mp;
        for(int i=0;i<arr.size();i++)
        mp[arr[i]]++;
        for(auto i:mp)
        {
            if(i.second>=2)
            ans.push_back(i.first);
        }
        
        return ans;
        
        
    }
};
