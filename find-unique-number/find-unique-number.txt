class Solution {
  public:
    int findUnique(vector<int> &arr) {
       unordered_map<int,int> mp;
       for(int i=0;i<arr.size();i++)
       {
           mp[arr[i]]++;
       }
       for(auto x:mp)
       {
           if(x.second==1)
           return x.first;
       }
       return -1;
    }
};
