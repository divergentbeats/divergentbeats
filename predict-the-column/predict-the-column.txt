class Solution {
  public:
    int maxZeros(vector<vector<int>> arr) {
        // code here
        int maxcount=0;
        int index=-1;
        for(int i=0;i<arr.size();i++)
        {
            int count=0;
            for(int j=0;j<arr[0].size();j++)
            {
                if(arr[j][i]==0)
                count++;
            }
            if(count>maxcount)
            {
                maxcount=count;
                index=i;
            }
        }
        
        return index;
    }
};
